let total = 0; function addExpense() { let name = document.getElementById("expenseName").value; let amount = Number(document.getElementById("expenseAmount").value); if(name === "" || amount <= 0){ alert("Enter valid data"); return; } let li = document.createElement("li"); li.textContent = `${name} - ₹${amount}`; document.getElementById("expenseList").appendChild(li); total += amount; document.getElementById("total").textContent = total; document.getElementById("expenseName").value = ""; document.getElementById("expenseAmount").value = ""; }
